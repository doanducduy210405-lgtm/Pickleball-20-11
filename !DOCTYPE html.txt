// --- 0. KHỞI TẠO KẾT NỐI BẢNG XẾP HẠNG TRỰC TUYẾN ---
const appId = typeof __app_id !== 'undefined' ? __app_id : 'pickleball-20-11-doan-duc-duy';
let db = null;
let auth = null;
let currentUser = null;
let leaderboardEntries = [];
let playerBestTotalScore = 0;
let playerHighestLevel = 0;
let levelScoresMap = {};

async function initCloudLeaderboard() {
    try {
        if (typeof __firebase_config === 'undefined') return;
        const firebaseConfig = JSON.parse(__firebase_config);
        const app = initializeApp(firebaseConfig);
        auth = getAuth(app);
        db = getFirestore(app);

        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
            await signInWithCustomToken(auth, __initial_auth_token);
        } else {
            await signInAnonymously(auth);
        }

        onAuthStateChanged(auth, (user) => {
            currentUser = user;
            if (user) {
                subscribeLeaderboard();
            }
        });
    } catch (err) {
        // Fallback chế độ nội bộ nếu không có cấu hình cloud
    }
}

function subscribeLeaderboard() {
    if (!db || !currentUser) return;
    const lbCol = collection(db, 'artifacts', appId, 'public', 'data', 'leaderboard');
    onSnapshot(lbCol, (snapshot) => {
        const list = [];
        snapshot.forEach((docSnap) => {
            const data = docSnap.data();
            if (data && typeof data.totalScore === 'number') {
                list.push({ id: docSnap.id, ...data });
            }
        });
        list.sort((a, b) => (b.totalScore - a.totalScore) || (b.highestLevel - a.highestLevel));
        leaderboardEntries = list;
        renderLeaderboardUI();
    }, () => {});
}

async function syncMyScoreToLeaderboard(completedLevelId, stageScore) {
    const prevBestForStage = levelScoresMap[completedLevelId] || 0;
    if (stageScore > prevBestForStage) {
        levelScoresMap[completedLevelId] = stageScore;
    }
    playerBestTotalScore = Object.values(levelScoresMap).reduce((sum, v) => sum + v, 0);
    if (completedLevelId > playerHighestLevel) {
        playerHighestLevel = completedLevelId;
    }

    const pNameInput = document.getElementById('input-player-name');
    const pName = (pNameInput && pNameInput.value.trim()) ? pNameInput.value.trim() : 'Tay Vợt Tri Ân';
    const ballCfg = getSelectedBallColor();

    updateMyLeaderboardHeader(pName);

    if (!db || !currentUser) {
        const existingIdx = leaderboardEntries.findIndex(e => e.id === 'local-me');
        const entry = {
            id: 'local-me',
            playerName: pName,
            totalScore: playerBestTotalScore,
            highestLevel: playerHighestLevel,
            ballColor: ballCfg.fill,
            updatedAt: Date.now()
        };
        if (existingIdx >= 0) leaderboardEntries[existingIdx] = entry;
        else leaderboardEntries.push(entry);
        leaderboardEntries.sort((a, b) => (b.totalScore - a.totalScore) || (b.highestLevel - a.highestLevel));
        renderLeaderboardUI();
        return;
    }

    try {
        const myDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'leaderboard', currentUser.uid);
        await setDoc(myDocRef, {
            playerName: pName,
            totalScore: playerBestTotalScore,
            highestLevel: playerHighestLevel,
            ballColor: ballCfg.fill,
            updatedAt: Date.now()
        }, { merge: true });
    } catch (e) {}
}

function updateMyLeaderboardHeader(pName) {
    document.getElementById('lb-my-name').textContent = pName;
    document.getElementById('lb-my-score').textContent = `${playerBestTotalScore.toLocaleString('vi-VN')} điểm`;
    document.getElementById('lb-my-stage').textContent = `Màn cao nhất: ${playerHighestLevel}/10`;
}

function renderLeaderboardUI() {
    const container = document.getElementById('leaderboard-list');
    if (!container) return;

    const pNameInput = document.getElementById('input-player-name');
    const currentName = (pNameInput && pNameInput.value.trim()) ? pNameInput.value.trim() : 'Tay Vợt Tri Ân';
    updateMyLeaderboardHeader(currentName);

    const samplePlayers = [
        { id: 'nph-duy', playerName: 'Đoàn Đức Duy (NPH)', totalScore: 61700, highestLevel: 10, ballColor: '#ccff00', isPublisher: true },
        { id: 'sample-1', playerName: 'Thầy Giáo Pickleball', totalScore: 53100, highestLevel: 9, ballColor: '#ff7b00' },
        { id: 'sample-2', playerName: 'Minh Anh - K65', totalScore: 36500, highestLevel: 7, ballColor: '#00e5ff' }
    ];

    const combined = [...leaderboardEntries];
    samplePlayers.forEach(sp => {
        if (!combined.some(item => item.id === sp.id)) {
            combined.push(sp);
        }
    });
    combined.sort((a, b) => (b.totalScore - a.totalScore) || (b.highestLevel - a.highestLevel));

    container.innerHTML = '';
    combined.slice(0, 30).forEach((entry, index) => {
        const rank = index + 1;
        const isMe = (currentUser && entry.id === currentUser.uid) || entry.id === 'local-me';
        const medal = rank === 1 ? '🥇' : rank === 2 ? '🥈' : rank === 3 ? '🥉' : `#${rank}`;

        const row = document.createElement('div');
        row.className = `flex items-center justify-between p-2.5 rounded-2xl border ${
            isMe
                ? 'bg-emerald-500/25 border-emerald-400 shadow'
                : rank <= 3
                    ? 'bg-white/10 border-amber-300/35'
                    : 'bg-white/5 border-white/10'
        }`;

        row.innerHTML = `
            <div class="flex items-center gap-2.5 min-w-0">
                <div class="w-7 h-7 rounded-xl bg-slate-900/90 flex items-center justify-center font-black text-xs shrink-0 ${
                    rank === 1 ? 'text-amber-300 text-base' : rank <= 3 ? 'text-amber-200 text-sm' : 'text-sky-200'
                }">
                    ${medal}
                </div>
                <span class="w-3 h-3 rounded-full shrink-0 border border-white/60" style="background:${entry.ballColor || '#ccff00'}"></span>
                <div class="min-w-0">
                    <div class="text-xs font-extrabold text-white truncate">
                        ${entry.playerName} ${isMe ? '<span class="text-[10px] text-emerald-300">(Bạn)</span>' : ''}
                    </div>
                    <div class="text-[10px] text-sky-200 font-semibold">
                        Đã chinh phục: Màn ${entry.highestLevel || 0}/10
                    </div>
                </div>
            </div>
            <div class="text-right shrink-0 ml-2">
                <div class="text-xs font-black text-amber-300">${(entry.totalScore || 0).toLocaleString('vi-VN')}đ</div>
            </div>
        `;
        container.appendChild(row);
    });
}

// --- 3. HỆ THỐNG ÂM THANH VUI NHỘN (WEB AUDIO API) ---
class SoundManager {
    constructor() {
        this.ctx = null;
        this.enabled = true;
    }
    init() {
        if (!this.ctx) {
            const AudioCtx = window.AudioContext || window.webkitAudioContext;
            if (AudioCtx) this.ctx = new AudioCtx();
        }
        if (this.ctx && this.ctx.state === 'suspended') {
            this.ctx.resume();
        }
    }
    playPop(pitchMult = 1) {
        if (!this.enabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(340 * pitchMult, now);
        osc.frequency.exponentialRampToValueAtTime(760 * pitchMult, now + 0.075);
        gain.gain.setValueAtTime(0.28, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.085);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.09);
    }
    playSweetHit() {
        if (!this.enabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        [523.25, 659.25, 783.99].forEach((freq, idx) => {
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(freq, now + idx * 0.03);
            gain.gain.setValueAtTime(0.18, now + idx * 0.03);
            gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.03 + 0.16);
            osc.connect(gain);
            gain.connect(this.ctx.destination);
            osc.start(now + idx * 0.03);
            osc.stop(now + idx * 0.03 + 0.17);
        });
    }
    playCollect(type) {
        if (!this.enabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        if (type === 'banana') {
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(280, now);
            osc.frequency.linearRampToValueAtTime(110, now + 0.25);
            gain.gain.setValueAtTime(0.18, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.26);
            osc.connect(gain);
            gain.connect(this.ctx.destination);
            osc.start(now);
            osc.stop(now + 0.27);
            return;
        }
        const notes = type === 'trophy' ? [587.33, 783.99, 1046.5] : [659.25, 987.77];
        notes.forEach((f, i) => {
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(f, now + i * 0.06);
            gain.gain.setValueAtTime(0.22, now + i * 0.06);
            gain.gain.exponentialRampToValueAtTime(0.001, now + i * 0.06 + 0.22);
            osc.connect(gain);
            gain.connect(this.ctx.destination);
            osc.start(now + i * 0.06);
            osc.stop(now + i * 0.06 + 0.23);
        });
    }
    playLevelClear() {
        if (!this.enabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        const melody = [523.25, 659.25, 783.99, 1046.5];
        melody.forEach((freq, idx) => {
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(freq, now + idx * 0.09);
            gain.gain.setValueAtTime(0.22, now + idx * 0.09);
            gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.09 + 0.35);
            osc.connect(gain);
            gain.connect(this.ctx.destination);
            osc.start(now + idx * 0.09);
            osc.stop(now + idx * 0.09 + 0.36);
        });
    }
    playVictoryFanfare() {
        if (!this.enabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        const melody = [523.25, 659.25, 783.99, 1046.5, 783.99, 1046.5];
        const times = [0, 0.12, 0.24, 0.36, 0.52, 0.68];
        melody.forEach((freq, idx) => {
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(freq, now + times[idx]);
            gain.gain.setValueAtTime(0.25, now + times[idx]);
            gain.gain.exponentialRampToValueAtTime(0.001, now + times[idx] + 0.4);
            osc.connect(gain);
            gain.connect(this.ctx.destination);
            osc.start(now + times[idx]);
            osc.stop(now + times[idx] + 0.42);
        });
    }
}

function createBall(isBonus = false) {
    const lvl = LEVELS[currentLevelIndex];

        trail: []
    };
}

function formatTimeMMSS(sec) {
    const m = Math.floor(sec / 60);
    const s = Math.floor(sec % 60);
    return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
}

    document.getElementById('hud-timer').textContent = `⏱️ ${formatTimeMMSS(levelElapsedSeconds)}`;

function addScore(points, x, y, customLabel = null) {
    const mult = doubleScoreTimer > 0 ? 2 : 1;
    const gained = points * mult;
    levelScore += gained;
    totalScoreAccumulated += gained;
    const label = customLabel || `+${gained}${mult > 1 ? ' (x2)' : ''}`;
    spawnFloatingText(x, y, label, mult > 1 ? '#fde047' : '#a7f3d0');
    updateHUD();

    const lvl = LEVELS[currentLevelIndex];
    if (levelScore >= lvl.targetScore && gameState === 'PLAYING') {
        handleLevelComplete();
    }
}

// --- 11. VÒNG LẶP CẬP NHẬT VẬT LÝ ---
function update(dt) {
    const lvl = LEVELS[currentLevelIndex];

    if (gameState === 'VICTORY') {
        confetti.forEach(c => {
            c.y += c.vy * dt;
            c.x += Math.sin(c.wobble) * 45 * dt;
            c.wobble += 4 * dt;
            c.rot += c.vRot * dt;
            if (c.y > height + 20) {
                c.y = -15;
                c.x = Math.random() * width;
            }
        });
        return;
    }

    if (gameState !== 'PLAYING') return;

    levelElapsedSeconds += dt;
    if (Math.floor(levelElapsedSeconds) !== Math.floor(levelElapsedSeconds - dt)) {
        document.getElementById('hud-timer').textContent = `⏱️ ${formatTimeMMSS(levelElapsedSeconds)}`;
    }

    if (doubleScoreTimer > 0) {
        doubleScoreTimer = Math.max(0, doubleScoreTimer - dt);
        if (doubleScoreTimer === 0) updateHUD();
    }
    if (slowPaddleTimer > 0) slowPaddleTimer = Math.max(0, slowPaddleTimer - dt);
    if (shakeTimer > 0) shakeTimer = Math.max(0, shakeTimer - dt);

    // Di chuyển vợt người chơi
    const lerpFactor = slowPaddleTimer > 0 ? 7.5 : 22;
    const prevX = player.x;
    player.x += (player.targetX - player.x) * Math.min(1, lerpFactor * dt);
    player.tilt = Math.max(-0.22, Math.min(0.22, (player.x - prevX) * 0.015));

    // AI đối thủ
    let leadBall = balls[0];
    balls.forEach(b => {
        if (b.vy < 0 && b.y < leadBall.y) leadBall = b;
    });
    if (leadBall) {
        const aiSpeed = (280 + lvl.speedMult * 95) * dt;
        const diff = leadBall.x - enemy.x;
        enemy.x += Math.sign(diff) * Math.min(Math.abs(diff), aiSpeed);
        const halfE = enemy.width / 2 + 10;
        enemy.x = Math.max(halfE, Math.min(width - halfE, enemy.x));
    }

    // Sinh vật phẩm
    spawnItemTimer -= dt;
    if (spawnItemTimer <= 0) {
        spawnItem();
        spawnItemTimer = lvl.spawnInterval + Math.random() * 0.85;
    }

    // Cập nhật bóng
    const ballCfg = getSelectedBallColor();
    for (let i = balls.length - 1; i >= 0; i--) {
        const b = balls[i];
        b.trail.push({ x: b.x, y: b.y });
        if (b.trail.length > 9) b.trail.shift();

        if (lvl.curveFactor > 0) {
            b.vx += Math.sin(b.y * 0.014) * lvl.curveFactor * 240 * dt;
        }

        b.x += b.vx * dt;
        b.y += b.vy * dt;

        // Nảy biên dọc
        if (b.x - b.radius < 14) {
            b.x = 14 + b.radius;
            b.vx = Math.abs(b.vx);
            audio.playPop(0.95);
        } else if (b.x + b.radius > width - 14) {
            b.x = width - 14 - b.radius;
            b.vx = -Math.abs(b.vx);
            audio.playPop(0.95);
        }

        // Va chạm vợt AI
        if (b.vy < 0 &&
            b.y - b.radius <= enemy.y + enemy.height / 2 &&
            b.y + b.radius >= enemy.y - enemy.height / 2 &&
            Math.abs(b.x - enemy.x) <= enemy.width / 2 + b.radius) {

            b.y = enemy.y + enemy.height / 2 + b.radius;
            const hitOffset = (b.x - enemy.x) / (enemy.width / 2);
            const bounceAngle = hitOffset * 0.72;
            const currentSpd = Math.min(b.speed * 1.15, Math.hypot(b.vx, b.vy) * 1.006);
            b.vx = Math.sin(bounceAngle) * currentSpd;
            b.vy = Math.cos(bounceAngle) * currentSpd;

            audio.playPop(0.85);
            spawnParticles(b.x, b.y, '#38bdf8', 7);
        }

        // Va chạm vợt Người chơi
        if (b.vy > 0 &&
            b.y + b.radius >= player.y - player.height / 2 &&
            b.y - b.radius <= player.y + player.height / 2 + 12 &&
            Math.abs(b.x - player.x) <= player.width / 2 + b.radius + 4) {

            b.y = player.y - player.height / 2 - b.radius;
            const hitOffset = (b.x - player.x) / (player.width / 2);
            const bounceAngle = Math.max(-0.82, Math.min(0.82, hitOffset * 0.78));
            const currentSpd = Math.min(b.speed * 1.18, Math.hypot(b.vx, b.vy) * 1.01);

            b.vx = Math.sin(bounceAngle) * currentSpd;
            b.vy = -Math.cos(bounceAngle) * currentSpd;

            combo++;
            const isSweetSpot = Math.abs(hitOffset) < 0.26;
            if (isSweetSpot) {
                audio.playSweetHit();
                shakeTimer = 0.08;
                spawnParticles(b.x, b.y, '#fde047', 15);
                const praise = PRAISE_WORDS[Math.floor(Math.random() * PRAISE_WORDS.length)];
                addScore(35 + Math.min(25, combo * 2), b.x, b.y - 24, `${praise} +${(35 + Math.min(25, combo * 2)) * (doubleScoreTimer > 0 ? 2 : 1)}`);
            } else {
                audio.playPop(1.12);
                spawnParticles(b.x, b.y, ballCfg.fill, 10);
                addScore(20 + Math.min(15, combo), b.x, b.y - 20);
            }
            updateHUD();
        }

        // Bóng lọt qua đáy
        if (b.y - b.radius > height + 10) {
            if (b.isBonus) {
                balls.splice(i, 1);
            } else {
                combo = 0;
                updateHUD();
                spawnFloatingText(Math.max(55, Math.min(width - 55, b.x)), height - 55, "Bóng mới! 🎾", "#bae6fd");
                b.x = width / 2;
                b.y = height * 0.34;
                b.trail = [];
                const ang = (Math.random() - 0.5) * 0.55;
                b.vx = Math.sin(ang) * b.speed * 0.85;
                b.vy = Math.cos(ang) * b.speed;
            }
        }
    }

    // Cập nhật vật phẩm rơi
    for (let i = items.length - 1; i >= 0; i--) {
        const it = items[i];
        it.y += it.vy * dt;
        it.wobble += 4.5 * dt;
        it.x += Math.sin(it.wobble) * 32 * dt;

        const dist = Math.hypot(it.x - player.x, it.y - player.y);
        if (dist <= player.width / 2 + it.radius + 6) {
            audio.playCollect(it.type);
            if (it.type === 'heart') {
                spawnParticles(it.x, it.y, '#fb7185', 14);
                addScore(65, it.x, it.y - 16, "❤️ +65");
            } else if (it.type === 'flower') {
                spawnParticles(it.x, it.y, '#f472b6', 16);
                addScore(110, it.x, it.y - 16, "💐 Hoa 20/11 +110");
            } else if (it.type === 'trophy') {
                doubleScoreTimer = 8.0;
                if (balls.length < 2) {
                    balls.push(createBall(true));
                }
                spawnParticles(it.x, it.y, '#facc15', 22);
                addScore(140, it.x, it.y - 20, "🏆 X2 ĐIỂM + ĐA BÓNG!");
            } else if (it.type === 'banana') {
                slowPaddleTimer = 2.0;
                shakeTimer = 0.14;
                spawnParticles(it.x, it.y, '#fef08a', 12);
                spawnFloatingText(it.x, it.y - 18, "🍌 Úi trơn quá!", "#fef08a");
            }
            items.splice(i, 1);
            continue;
        }

        if (it.y > height + 30) {
            items.splice(i, 1);
        }
    }

    // Cập nhật hạt & chữ bay
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.life -= dt;
        if (p.life <= 0) particles.splice(i, 1);
    }
    for (let i = floatingTexts.length - 1; i >= 0; i--) {
        const ft = floatingTexts[i];
        ft.y -= 48 * dt;
        ft.life -= dt;
        if (ft.life <= 0) floatingTexts.splice(i, 1);
    }
}

// --- 12. VẼ SÂN ĐẤU & NHÂN VẬT ---
function drawCourt() {
    const lvl = LEVELS[currentLevelIndex];
    const pad = 14;
    const courtW = width - pad * 2;
    const courtH = height - pad * 2;

    ctx.fillStyle = lvl.courtOuter;
    ctx.fillRect(0, 0, width, height);

    ctx.fillStyle = lvl.courtInner;
    ctx.beginPath();
    ctx.roundRect(pad, pad, courtW, courtH, 18);
    ctx.fill();

    const netY = height * 0.48;
    const kitchenH = height * 0.14;

    ctx.fillStyle = lvl.kitchenColor;
    ctx.fillRect(pad, netY - kitchenH, courtW, kitchenH * 2);

    ctx.strokeStyle = 'rgba(255, 255, 255, 0.85)';
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.roundRect(pad, pad, courtW, courtH, 18);
    ctx.stroke();

    ctx.beginPath();
    ctx.moveTo(pad, netY - kitchenH);
    ctx.lineTo(width - pad, netY - kitchenH);
    ctx.moveTo(pad, netY + kitchenH);
    ctx.lineTo(width - pad, netY + kitchenH);
    ctx.stroke();

    ctx.beginPath();
    ctx.moveTo(width / 2, pad);
    ctx.lineTo(width / 2, netY - kitchenH);
    ctx.moveTo(width / 2, netY + kitchenH);
    ctx.lineTo(width / 2, height - pad);
    ctx.stroke();

    ctx.fillStyle = 'rgba(255, 255, 255, 0.26)';
    ctx.font = '800 11px "Be Vietnam Pro", sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('KITCHEN • NON-VOLLEY ZONE', width / 2, netY + kitchenH - 8);
    ctx.fillText(`MÀN ${lvl.id}: ${lvl.name.toUpperCase()} • CHÀO MỪNG 20/11`, width / 2, netY - kitchenH - 10);

    // Lưới Pickleball
    ctx.save();
    ctx.fillStyle = 'rgba(15, 23, 42, 0.45)';
    ctx.fillRect(pad - 2, netY - 5, courtW + 4, 10);
    ctx.strokeStyle = 'rgba(255,255,255,0.45)';
    ctx.lineWidth = 1;
    for (let x = pad; x < width - pad; x += 10) {
        ctx.beginPath();
        ctx.moveTo(x, netY - 5);
        ctx.lineTo(x, netY + 5);
        ctx.stroke();
    }
    ctx.fillStyle = '#ffffff';
    ctx.fillRect(pad - 3, netY - 2, courtW + 6, 4);
    ctx.restore();
}

function drawEnemy() {

    ctx.save();

function drawPlayer() {
    ctx.save();
    ctx.translate(player.x, player.y);
    ctx.rotate(player.tilt);

    ctx.fillStyle = 'rgba(0,0,0,0.34)';
    ctx.beginPath();
    ctx.ellipse(0, 14, player.width / 2, 8, 0, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = '#78350f';
    ctx.beginPath();
    ctx.roundRect(-7, player.height / 2 - 2, 14, 18, 5);
    ctx.fill();

    const grad = ctx.createLinearGradient(-player.width / 2, 0, player.width / 2, 0);
    if (slowPaddleTimer > 0) {
        grad.addColorStop(0, '#fef08a');
        grad.addColorStop(1, '#eab308');
    } else if (doubleScoreTimer > 0) {
        grad.addColorStop(0, '#fde047');
        grad.addColorStop(0.5, '#fb923c');
        grad.addColorStop(1, '#f43f5e');
    } else {
        grad.addColorStop(0, '#fb923c');
        grad.addColorStop(0.5, '#f97316');
        grad.addColorStop(1, '#ea580c');
    }

    ctx.fillStyle = grad;
    ctx.strokeStyle = '#ffffff';
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.roundRect(-player.width / 2, -player.height / 2, player.width, player.height, 10);
    ctx.fill();
    ctx.stroke();

    ctx.fillStyle = 'rgba(255, 255, 255, 0.48)';
    ctx.beginPath();
    ctx.roundRect(-player.width * 0.14, -4, player.width * 0.28, 8, 4);
    ctx.fill();

    ctx.restore();
}

function drawBalls() {
    const ballCfg = getSelectedBallColor();
    balls.forEach(b => {
        b.trail.forEach((pt, idx) => {
            const ratio = (idx + 1) / b.trail.length;
            ctx.fillStyle = b.isBonus ? `rgba(251, 191, 36, ${ratio * 0.35})` : ballCfg.trail;
            ctx.beginPath();
            ctx.arc(pt.x, pt.y, b.radius * ratio * 0.88, 0, Math.PI * 2);
            ctx.fill();
        });

        ctx.save();
        ctx.translate(b.x, b.y);
        ctx.fillStyle = b.isBonus ? '#fbbf24' : ballCfg.fill;
        ctx.strokeStyle = b.isBonus ? '#b45309' : ballCfg.stroke;
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.arc(0, 0, b.radius, 0, Math.PI * 2);
        ctx.fill();
        ctx.stroke();

        ctx.fillStyle = b.isBonus ? '#78350f' : ballCfg.hole;
        const holes = [[-4, -4], [4, -4], [0, 0], [-4, 4], [4, 4]];
        holes.forEach(([hx, hy]) => {
            ctx.beginPath();
            ctx.arc(hx, hy, 1.8, 0, Math.PI * 2);
            ctx.fill();
        });
        ctx.restore();
    });
}

function drawItemsAndFX() {
    items.forEach(it => {
        ctx.save();
        ctx.translate(it.x, it.y);
        ctx.fillStyle = it.type === 'banana' ? 'rgba(254, 240, 138, 0.3)' : 'rgba(255, 255, 255, 0.26)';
        ctx.beginPath();
        ctx.arc(0, 0, it.radius + 3, 0, Math.PI * 2);
        ctx.fill();

        ctx.font = '24px sans-serif';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText(it.emoji, 0, 1);
        ctx.restore();
    });

    particles.forEach(p => {
        ctx.globalAlpha = Math.max(0, p.life / p.maxLife);
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
    });
    ctx.globalAlpha = 1;

    floatingTexts.forEach(ft => {
        ctx.save();
        ctx.globalAlpha = Math.max(0, ft.life / ft.maxLife);
        ctx.font = '900 15px "Be Vietnam Pro", sans-serif';
        ctx.textAlign = 'center';
        ctx.strokeStyle = 'rgba(15, 23, 42, 0.9)';
        ctx.lineWidth = 4;
        ctx.strokeText(ft.text, ft.x, ft.y);
        ctx.fillStyle = ft.color;
        ctx.fillText(ft.text, ft.x, ft.y);
        ctx.restore();
    });

    if (gameState === 'VICTORY') {
        confetti.forEach(c => {
            ctx.save();
            ctx.translate(c.x, c.y);
            ctx.rotate(c.rot);
            ctx.fillStyle = c.color;
            ctx.fillRect(-c.w / 2, -c.h / 2, c.w, c.h);
            ctx.restore();
        });
    }
}

function render() {
    if (!screenStart.classList.contains('hidden')) {
        drawBallPreview();
    }

    ctx.save();
    if (shakeTimer > 0) {
        const dx = (Math.random() - 0.5) * 7;
        const dy = (Math.random() - 0.5) * 7;
        ctx.translate(dx, dy);
    }

    drawCourt();
    drawEnemy();
    drawPlayer();
    if (gameState === 'PLAYING' || gameState === 'PAUSED') {
        drawBalls();
    }
    drawItemsAndFX();
    ctx.restore();
}

let lastTime = performance.now();
function gameLoop(now = performance.now()) {
    const dt = Math.min(0.05, (now - lastTime) / 1000);
    lastTime = now;

    update(dt);
    render();
    requestAnimationFrame(gameLoop);
}

// --- 14. XỬ LÝ QUA MÀN (LEVEL CLEAR) & CHIẾN THẮNG CHUNG CUỘC 10 MÀN ---
function handleLevelComplete() {
    const lvl = LEVELS[currentLevelIndex];
    completedLevels.add(lvl.id);
    syncMyScoreToLeaderboard(lvl.id, levelScore);

    if (currentLevelIndex >= LEVELS.length - 1) {
        triggerGrandVictory();
        return;
    }

    gameState = 'LEVEL_CLEAR';
    audio.playLevelClear();

    hudBar.classList.add('hidden');
    progressContainer.classList.add('hidden');
    bottomHint.classList.add('hidden');

    document.getElementById('clear-level-num').textContent = lvl.id;
    document.getElementById('clear-level-title').textContent = `Hoàn Thành Màn ${lvl.id}: ${lvl.name}!`;
    document.getElementById('clear-stage-score').textContent = levelScore.toLocaleString('vi-VN');
    document.getElementById('clear-stage-time').textContent = formatTimeMMSS(levelElapsedSeconds);
    document.getElementById('clear-piece-num').textContent = lvl.id;
    document.getElementById('clear-unlocked-wish').textContent = `"${lvl.wish}"`;

    const nextLvl = LEVELS[currentLevelIndex + 1];
    document.getElementById('btn-next-level').textContent = `TIẾP TỤC MÀN ${nextLvl.id}: ${nextLvl.name.toUpperCase()} ➡️`;

    screenLevelClear.classList.remove('hidden');
}

function triggerGrandVictory() {
    gameState = 'VICTORY';
    audio.playVictoryFanfare();

    const tInput = document.getElementById('input-teacher-name').value.trim();
    const sInput = document.getElementById('input-sender-name').value.trim();
    if (tInput) teacherName = tInput;
    if (sInput) senderName = sInput;

    hudBar.classList.add('hidden');
    progressContainer.classList.add('hidden');
    bottomHint.classList.add('hidden');
    screenStart.classList.add('hidden');
    screenLevelClear.classList.add('hidden');

    document.getElementById('card-teacher-name').textContent = teacherName;
    document.getElementById('card-sender-name').textContent = senderName;
    document.getElementById('card-total-score').textContent = Math.max(totalScoreAccumulated, levelScore).toLocaleString('vi-VN');
    document.getElementById('snapshot-toast').classList.add('hidden');

    const colors = ['#facc15', '#fb7185', '#38bdf8', '#4ade80', '#c084fc', '#fb923c'];
    confetti = [];
    for (let i = 0; i < 85; i++) {
        confetti.push({
            x: Math.random() * width,
            y: Math.random() * height - height,
            vy: 95 + Math.random() * 145,
            w: 7 + Math.random() * 6,
            h: 4 + Math.random() * 5,
            color: colors[i % colors.length],
            rot: Math.random() * Math.PI * 2,
            vRot: (Math.random() - 0.5) * 8,
            wobble: Math.random() * Math.PI * 2
        });
    }

    screenVictory.classList.remove('hidden');
}
